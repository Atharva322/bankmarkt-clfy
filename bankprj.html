# 1) Read data
bank <- read.csv("bank-additional.csv", sep = ";")

# 2) Response as factor with fixed levels
bank$y <- factor(bank$y, levels = c("no", "yes"))

# 3) Drop duration and default (to avoid factor-level headaches)
bank <- bank %>%
  select(-duration, -default)

# 4) Make other categorical variables factors
bank <- bank %>%
  mutate(
    job         = factor(job),
    marital     = factor(marital),
    education   = factor(education),
    housing     = factor(housing),
    loan        = factor(loan),
    contact     = factor(contact),
    month       = factor(month),
    day_of_week = factor(day_of_week),
    poutcome    = factor(poutcome)
  )

# 5) Train–test split
set.seed(123)
n <- nrow(bank)
train_idx <- sample(1:n, 0.7 * n)
train <- bank[train_idx, ]
test  <- bank[-train_idx, ]

Bootstrap for stability (AUC of logistic regression)

logit_boot_auc <- function(data, indices) {
  # Bootstrap sample
  d <- data[indices, ]
  
  # Fit logistic regression on bootstrap sample
  fit <- glm(y ~ ., data = d, family = binomial)
  
  # Predict on the same fixed test set to keep comparison fair
  probs <- predict(fit, newdata = test, type = "response")
  
  # Return AUC as a single number
  auc(roc(test$y, probs))
}
